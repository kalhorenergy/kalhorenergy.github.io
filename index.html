(function() { "use strict"; const toggle = document.getElementById("languageToggle"); const dropdown = document.getElementById("languageDropdown"); const search = document.getElementById("languageSearch"); const optionsBox = document.getElementById("languageOptions"); const status = document.getElementById("languageStatus"); const host = document.getElementById("google_translate_element"); const rtlLanguages = [ "ar","arc","az","ckb","dv","fa","he","ku", "nqo","ps","sd","ug","ur","yi" ]; let languages = []; let selected = "en"; let timer = null; function getSavedLanguage() { const match = document.cookie.match( /(?:^|;\s*)googtrans=([^;]+)/ ); if (!match) return "en"; try { const parts = decodeURIComponent(match[1]).split("/"); return parts.length >= 3 ? parts[2] || "en" : "en"; } catch(e) { return "en"; } } function saveLanguage(code) { if (code === "en") { document.cookie = "googtrans=; expires=Thu, 01 Jan 1970 00:00:00 GMT; path=/; SameSite=Lax"; } else { document.cookie = "googtrans=" + encodeURIComponent("/en/" + code) + "; path=/; max-age=31536000; SameSite=Lax"; } } function updateDirection(code) { const lang = (code || "en").toLowerCase().split("-")[0]; document.documentElement.dir = rtlLanguages.includes(lang) ? "rtl" : "ltr"; } function getGoogleSelect() { return host.querySelector("select.goog-te-combo") || document.querySelector("select.goog-te-combo"); } function setStatus(message) { status.textContent = message; } function renderLanguages(query) { optionsBox.replaceChildren(); const term = (query || "").trim().toLowerCase(); const filtered = languages.filter(item => item.name.toLowerCase().includes(term) || item.code.toLowerCase().includes(term) ); if (!filtered.length) { const empty = document.createElement("div"); empty.className = "language-empty"; empty.textContent = "No matching language found."; optionsBox.appendChild(empty); return; } filtered.forEach(item => { const button = document.createElement("button"); button.type = "button"; button.className = "language-option"; button.textContent = item.name; if (item.code === selected) { button.classList.add("active"); button.setAttribute("aria-selected", "true"); } button.addEventListener("click", () => { selectLanguage(item.code); }); optionsBox.appendChild(button); }); } function loadLanguages() { const select = getGoogleSelect(); if (!select || !select.options.length) return false; const found = []; Array.from(select.options).forEach(option => { const code = option.value; const name = option.textContent.trim(); if (!name) return; if (!found.some(item => item.code === code)) { found.push({ code: code || "en", name: code ? name : "English (Original)" }); } }); if (!found.length) return false; languages = found; selected = getSavedLanguage(); if (!languages.some(item => item.code === selected)) { selected = "en"; } updateDirection(selected); renderLanguages(search.value); const current = languages.find(item => item.code === selected); toggle.textContent = "🌐 " + (current ? current.name : "Language") + " ▾"; setStatus(languages.length + " available language options"); return true; } function selectLanguage(code) { const select = getGoogleSelect(); if (!select) { setStatus("Google Translate is not ready. Please try again."); startLoading(); return; } const googleCode = code === "en" ? "" : code; const available = Array.from(select.options).some( option => option.value === googleCode ); if (!available) { setStatus("This language is not currently available."); return; } selected = code; saveLanguage(code); updateDirection(code); select.value = googleCode; select.dispatchEvent( new Event("change", { bubbles: true }) ); renderLanguages(search.value); const current = languages.find(item => item.code === code); toggle.textContent = "🌐 " + (current ? current.name : "Language") + " ▾"; setStatus("Language selected."); closeMenu(); } function openMenu() { dropdown.classList.add("open"); toggle.setAttribute("aria-expanded", "true"); if (!loadLanguages()) { setStatus("Loading available languages..."); startLoading(); } search.focus(); } function closeMenu() { dropdown.classList.remove("open"); toggle.setAttribute("aria-expanded", "false"); } function startLoading() { if (timer) return; let attempts = 0; timer = setInterval(() => { attempts++; if (loadLanguages() || attempts >= 40) { clearInterval(timer); timer = null; if (!languages.length) { setStatus( "Google Translate could not be loaded. Check your connection." ); } } }, 500); } toggle.addEventListener("click", event => { event.stopPropagation(); if (dropdown.classList.contains("open")) { closeMenu(); } else { openMenu(); } }); search.addEventListener("input", () => { renderLanguages(search.value); }); search.addEventListener("keydown", event => { if (event.key === "Escape") { closeMenu(); toggle.focus(); } if (event.key === "Enter") { const first = optionsBox.querySelector(".language-option"); if (first) first.click(); } }); document.addEventListener("click", event => { if (!event.target.closest(".language-box")) { closeMenu(); } }); document.addEventListener("keydown", event => { if (event.key === "Escape") closeMenu(); }); updateDirection(getSavedLanguage()); const observer = new MutationObserver(() => { if (loadLanguages()) { if (timer) { clearInterval(timer); timer = null; } } }); observer.observe(host, { childList: true, subtree: true }); startLoading(); })();
